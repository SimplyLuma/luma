# Upstream libpiper with distro-owned ONNX/espeak. No build-time downloader.
cmake_minimum_required(VERSION 3.26)
project(luma_piper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ESPEAK REQUIRED IMPORTED_TARGET espeak-ng)
find_path(ONNX_INCLUDE onnxruntime_cxx_api.h PATH_SUFFIXES onnxruntime onnxruntime/core/session REQUIRED)
find_library(ONNX_LIBRARY onnxruntime REQUIRED)
set(PIPER_SOURCE_DIR "" CACHE PATH "Exact unpacked Piper libpiper source")
if(NOT EXISTS "${PIPER_SOURCE_DIR}/src/piper.cpp")
 message(FATAL_ERROR "Pinned Piper source is required")
endif()
add_library(luma-piper SHARED ${PIPER_SOURCE_DIR}/src/piper.cpp)
target_include_directories(luma-piper PUBLIC ${PIPER_SOURCE_DIR}/include ${ONNX_INCLUDE})
target_link_libraries(luma-piper PRIVATE PkgConfig::ESPEAK ${ONNX_LIBRARY})
set_target_properties(luma-piper PROPERTIES VERSION 1.4.2 SOVERSION 1)
install(TARGETS luma-piper LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
add_executable(synthesis-check tests/synthesis.cpp)
target_link_libraries(synthesis-check PRIVATE luma-piper)
