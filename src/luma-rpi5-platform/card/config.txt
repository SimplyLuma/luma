# Luma for Raspberry Pi 5
# Board firmware -> U-Boot (EFI) -> GRUB (EFI/BOOT/grub.cfg) -> Luma.
# Luma's Raspberry Pi kernel (kernel-luma-rpi) runs on the device tree the
# board firmware builds from the .dtb files and overlays/ on this partition,
# so dtoverlay= and dtparam= lines below work as on Raspberry Pi OS. U-Boot and
# GRUB pass that tree on unchanged (there is no dtb/ directory here).
# luma-rpi5-firmware-sync refreshes the .dtb files and overlays/ from the
# running kernel after a good boot. Nothing in Luma rewrites this file.

[pi5]
arm_64bit=1
kernel=u-boot.bin
arm_boost=1
enable_uart=1
disable_overscan=1
disable_fw_kms_setup=1
hdmi_ignore_cec_init=1
os_check=0
# KMS display and HDMI audio, 256 MB CMA below 1 GB.
dtoverlay=vc4-kms-v3d-pi5,cma-256
# The PCIe connector (NVMe, HATs); the firmware only turns it on by itself
# when it boots from NVMe.
dtparam=pciex1

[all]
